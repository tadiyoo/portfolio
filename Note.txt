<script>
  document.addEventListener('DOMContentLoaded', async () => {
      const pageIdentifier = "home"; // Change this to match your page
      const counterElement = document.getElementById('visitorCount');
      
      try {
          // Show loading state
          counterElement.textContent = '...';
          
          // Debug: Show the API URL being called
          const apiUrl = `https://coachingappapi.arslanishaq.com/api/visit-counter?page_url=${pageIdentifier}`;
          // alert(`Attempting to call API:\n${apiUrl}`);
          
          // Call your API
          const response = await fetch(apiUrl);
          
          // // Debug: Show raw response
          // alert(`API Response Status: ${response.status}\n${response.statusText}`);
          
          if (!response.ok) {
              throw new Error(`HTTP error! status: ${response.status}`);
          }
          
          // Get and display the count
          const count = await response.text();
          // alert(`Received count value: ${count}`);
          counterElement.textContent = count;
          
      } catch (error) {
          // alert(`Error occurred:\n${error.message}`);
          counterElement.textContent = '0';
      }
  });
</script>
